```kotlin
package com.allday.detoxy.domain.model

data class TimeSlot(
    val startHour: Int,          // 0-23
    val startMinute: Int,        // 0-59
    val durationMinutes: Int,    // 15-720 (15분~12시간)
    val presetType: String       // "STANDARD", "MEDIUM", "COMPLETE"
) {
    /**
     * 포맷팅된 시작 시간 (예: "10:00")
     */
    fun formatStartTime(): String {
        return String.format("%02d:%02d", startHour, startMinute)
    }
    
    /**
     * 포맷팅된 기간 (예: "1시간 30분")
     */
    fun formatDuration(): String {
        val hours = durationMinutes / 60
        val minutes = durationMinutes % 60
        
        return when {
            hours > 0 && minutes > 0 -> "${hours}시간 ${minutes}분"
            hours > 0 -> "${hours}시간"
            else -> "${minutes}분"
        }
    }
    
    companion object {
        const val MIN_DURATION = 15
        const val MAX_DURATION = 720
        const val DEFAULT_DURATION = 90
    }
}
```

```kotlin
@HiltViewModel
class ScheduleGroupViewModel @Inject constructor(
    private val repository: ScheduleGroupRepository,
    private val timeBasedRepository: TimeBasedAutoRunRepository,  // 🆕 주입
    private val scheduleManager: ScheduleGroupManager
) : ViewModel() {
    
    // ... 기존 메서드들
    
    /**
     * 🆕 시간대를 포함한 시간표 생성 (Phase 1)
     */
    suspend fun createScheduleGroupWithTimeSlots(
        name: String,
        description: String?,
        timeSlots: List<TimeSlot>
    ): String {
        if (name.isBlank()) {
            _errorState.value = "그룹 이름을 입력해주세요"
            throw IllegalArgumentException("그룹 이름을 입력해주세요")
        }
        
        if (timeSlots.isEmpty()) {
            _errorState.value = "최소 1개 이상의 시간대를 추가해주세요"
            throw IllegalArgumentException("최소 1개 이상의 시간대를 추가해주세요")
        }
        
        try {
            _isLoading.value = true
            
            // 1. ScheduleGroup 생성
            val scheduleGroup = ScheduleGroup(
                name = name.trim(),
                description = description?.trim()?.takeIf { it.isNotEmpty() }
            )
            repository.insert(scheduleGroup)
            
            // 2. TimeBasedAutoRun 생성 (각 시간대마다)
            timeSlots.forEach { slot ->
                val autoRun = TimeBasedAutoRun(
                    hour = slot.startHour,
                    minute = slot.startMinute,
                    durationMinutes = slot.durationMinutes,
                    presetType = slot.presetType,
                    scheduleGroupId = scheduleGroup.id,  // 🆕 그룹 연결
                    isIndependent = false,  // 그룹에 종속
                    isEnabled = true,
                    createdAt = System.currentTimeMillis()
                )
                timeBasedRepository.insert(autoRun)
            }
            
            return scheduleGroup.id
        } catch (e: Exception) {
            _errorState.value = "그룹 생성 실패: ${e.message}"
            throw e
        } finally {
            _isLoading.value = false
        }
    }
}
```

### 3.1 LocationConflictResolver 구현 (Day 11-12)

**파일**: `app/src/main/java/com/allday/detoxy/core/manager/LocationConflictResolver.kt` (신규)

- [ ] LocationConflictResolver 클래스 생성 (~300줄)

```kotlin
package com.allday.detoxy.core.manager

import android.content.Context
import android.location.Location
import android.util.Log
import com.allday.detoxy.data.local.dao.LocationBasedAutoRunDao
import com.allday.detoxy.data.local.entity.LocationBasedAutoRun
import dagger.hilt.android.qualifiers.ApplicationContext
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class LocationConflictResolver @Inject constructor(
    private val locationDao: LocationBasedAutoRunDao,
    @ApplicationContext private val context: Context
) {
    companion object {
        private const val TAG = "ConflictResolver"
        private const val HYSTERESIS_SECONDS = 120  // 2분
    }
    
    private var lastActiveLocationId: String? = null
    private var lastActivationTime: Long = 0
    
    /**
     * 위치 충돌 해소 규칙 (기반: 스케줄가동프로세스.md):
     * 1. 면적(반경) 우선
     * 2. 최근 활성 유지 (히스테리시스 120초)
     *    - 새 위치 전환 시 타이머 리셋
     *    - 만료 시 즉시 재평가
     * 3. 거리 우선
     * 4. 최근 수정(updatedAt) 우선
     */
    suspend fun resolveConflict(
        currentLocation: Location,
        candidates: List<LocationBasedAutoRun>
    ): LocationBasedAutoRun? {
        
        if (candidates.isEmpty()) {
            Log.d(TAG, "No candidates")
            return null
        }
        
        if (candidates.size == 1) {
            Log.d(TAG, "Single candidate: ${candidates.first().label}")
            return candidates.first()
        }
        
        Log.d(TAG, "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
        Log.d(TAG, "🔍 Conflict Resolution Start")
        
        // 1. 면적 우선: 반경이 가장 작은 위치들만 남김
        val minRadius = candidates.minOf { it.radiusMeters }
        var filtered = candidates.filter { it.radiusMeters == minRadius }
        
        Log.d(TAG, "📐 Step 1: Area Priority (min radius: ${minRadius}m)")
        
        if (filtered.size == 1) {
            Log.d(TAG, "✅ Winner: ${filtered.first().label} (area)")
            return filtered.first()
        }
        
        // 2. 히스테리시스: 직전 활성 위치가 후보에 포함되고, 120초 이내면 유지
        val lastActive = lastActiveLocationId
        val timeSinceActivation = (System.currentTimeMillis() - lastActivationTime) / 1000
        
        Log.d(TAG, "⏱️ Step 2: Hysteresis")
        Log.d(TAG, "  Last active: $lastActive, Time since: ${timeSinceActivation}s")
        
        if (lastActive != null && timeSinceActivation <= HYSTERESIS_SECONDS) {
            val lastActiveInCandidates = filtered.find { it.id == lastActive }
            if (lastActiveInCandidates != null) {
                Log.d(TAG, "✅ Winner: ${lastActiveInCandidates.label} (hysteresis)")
                return lastActiveInCandidates
            } else {
                Log.d(TAG, "  ⚠️ Last active not in candidates → reset hysteresis")
                resetHysteresis()
            }
        } else if (timeSinceActivation > HYSTERESIS_SECONDS) {
            Log.d(TAG, "  ⚠️ Hysteresis expired → reset and re-evaluate")
            resetHysteresis()
        }
        
        // 3. 거리 우선: 가장 가까운 위치
        val distances = filtered.map { location ->
            val targetLocation = Location("").apply {
                latitude = location.latitude
                longitude = location.longitude
            }
            val distance = currentLocation.distanceTo(targetLocation)
            location to distance
        }
        
        Log.d(TAG, "📏 Step 3: Distance Priority")
        distances.forEach { (loc, dist) ->
            Log.d(TAG, "  - ${loc.label}: ${String.format("%.1f", dist)}m")
        }
        
        val minDistance = distances.minOf { it.second }
        filtered = distances.filter { it.second == minDistance }.map { it.first }
        
        if (filtered.size == 1) {
            Log.d(TAG, "✅ Winner: ${filtered.first().label} (distance)")
            return filtered.first()
        }
        
        // 4. 최근 수정(updatedAt) 우선
        // Note: updatedAt 필드는 향후 추가 예정, 현재는 createdAt 사용
        val winner = filtered.maxByOrNull { it.createdAt }  // TODO: updatedAt으로 변경
        
        Log.d(TAG, "🕐 Step 4: Recent Modification Priority")
        Log.d(TAG, "✅ Winner: ${winner?.label} (recent)")
        Log.d(TAG, "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
        
        return winner
    }
    
    /**
     * 활성화 기록 (히스테리시스용)
     * 새 위치로 전환 시 호출하여 타이머 리셋
     */
    fun recordActivation(locationId: String) {
        val previousLocation = lastActiveLocationId
        lastActiveLocationId = locationId
        lastActivationTime = System.currentTimeMillis()
        
        if (previousLocation != locationId) {
            Log.d(TAG, "📝 Location switched: $previousLocation → $locationId (timer reset)")
        } else {
            Log.d(TAG, "📝 Same location re-activated: $locationId")
        }
    }
    
    /**
     * 히스테리시스 상태 초기화
     * - 직전 활성 위치가 후보에서 제외됨
     * - 히스테리시스 시간 만료
     */
    fun resetHysteresis() {
        lastActiveLocationId = null
        lastActivationTime = 0
        Log.d(TAG, "🔄 Hysteresis reset")
    }
}
```

```kotlin
     * 🆕 위치 기반 활성화 (충돌 해소 포함) (Phase 3)
     */
    suspend fun activateGroupByLocation(
        currentLocation: android.location.Location,
        candidates: List<LocationBasedAutoRun>
    ): Result<Unit> = runCatching {
        Log.i(TAG, "🌍 Activating group by location")
        
        // 충돌 해소
        val winner = conflictResolver.resolveConflict(currentLocation, candidates)
            ?: throw IllegalStateException("No suitable location found")
        
        val scheduleGroupId = winner.linkedScheduleGroupId
            ?: throw IllegalStateException("Location has no linked schedule group: ${winner.label}")
        
        // 시간표 활성화
        activateGroup(scheduleGroupId).getOrThrow()
        
        // 활성화 기록 (히스테리시스용)
        conflictResolver.recordActivation(winner.id)
        
        Log.i(TAG, "✅ Schedule group activated: $scheduleGroupId (location: ${winner.label})")
    }
}
```

```kotlin
    val userLocation = geofencingEvent.triggeringLocation ?: return  // 🆕 사용자 위치

            val locationIds = triggeringGeofences.map { it.requestId }
            val candidates = mutableListOf<LocationBasedAutoRun>()
            
            locationIds.forEach { id ->
                val location = locationDao.getByIdOnce(id)
                if (location != null && location.activateScheduleOnEnter) {
                    candidates.add(location)
                }
            }
            
            if (candidates.isNotEmpty()) {
                // 🆕 충돌 해소 후 시간표 활성화
                val result = scheduleManager.activateGroupByLocation(
                    currentLocation = userLocation,
                    candidates = candidates

                } else {
                    Log.e(TAG, "⚠️ Failed to activate schedule", result.exceptionOrNull())
                }
            }
        } catch (e: Exception) {
            Log.e(TAG, "⚠️ Error handling geofence enter", e)
        } finally {
            pendingResult.finish()
        }
    }
```

```kotlin
     *
     * 우선순위 (기반: 스케쥴가동프로세스.md):
     * 1. 더 구체적인 스케줄 우선 (평일 > 매일)
     * 2. 최근 수정(updatedAt) 우선
     * 3. 생성 순서(createdAt) 우선
     */
    suspend fun activateDefaultPolicy(): Result<Unit> = runCatching {
        // 1. 모든 스케줄 그룹 조회
        val allGroups = repository.getAll().first()

        // 2. 비위치 스케줄만 필터링 (linkedScheduleGroupId == null)
        val nonLocationGroups = allGroups.filter { group ->
            val linkedLocations = repository.getLinkedLocations(group.id)
            linkedLocations.isEmpty()
        }

        if (nonLocationGroups.isEmpty()) {
            Log.d(TAG, "No default policy available")
            return Result.success(Unit)
        }

        // 3. 현재 시각/요일에 맞는 스케줄 필터링
        val now = LocalDateTime.now()
        val currentDay = now.dayOfWeek
        val currentTime = now.toLocalTime()

        val applicableCandidates = nonLocationGroups.filter { group ->
            val timeSlots = repository.getLinkedTimeBasedAutoRuns(group.id)
            timeSlots.any { slot ->
                // 요일 체크
                val daysEnabled = slot.enabledDays.contains(currentDay)

                // 시간대 체크
                val slotStart = LocalTime.of(slot.hour, slot.minute)
                val slotEnd = slotStart.plusMinutes(slot.durationMinutes.toLong())
                val inTimeRange = currentTime in slotStart..slotEnd

                daysEnabled && inTimeRange
            }
        }

        if (applicableCandidates.isEmpty()) {
            Log.d(TAG, "No applicable default policy for current time/day")
            return Result.success(Unit)
        }

        // 4. 우선순위 적용 (2개 이상일 때)
        val selectedPolicy = if (applicableCandidates.size == 1) {
            applicableCandidates.first()
        } else {
            Log.d(TAG, "Multiple applicable schedules: ${applicableCandidates.size}")
            selectByPriority(applicableCandidates)
        }

        // 5. 기본 정책 적용
        repository.toggleActiveWithTimestamp(selectedPolicy.id, true, System.currentTimeMillis())

        // 6. 알람 등록
        val timeSlots = repository.getLinkedTimeBasedAutoRuns(selectedPolicy.id)
        timeSlots.filter { it.isEnabled }.forEach { slot ->
            alarmManager.scheduleTimeBasedAutoRun(slot)
        }

        Log.i(TAG, "✅ Default policy activated: ${selectedPolicy.name}")
    }

    /**
     * 비위치 스케줄 우선순위 선택
     * 1. 더 구체적인 스케줄 (평일 > 매일, 주말 > 매일)
     * 2. 최근 수정(updatedAt) 우선 (TODO: updatedAt 필드 추가 시)
     * 3. 생성 순서(createdAt) 우선
     */
    private suspend fun selectByPriority(candidates: List<ScheduleGroup>): ScheduleGroup {
        // 1. 구체성 점수 계산
        val scored = candidates.map { group ->
            val timeSlots = repository.getLinkedTimeBasedAutoRuns(group.id).first()
            val specificityScore = calculateSpecificityScore(timeSlots)
            group to specificityScore
        }

        Log.d(TAG, "Specificity scores:")
        scored.forEach { (group, score) ->
            Log.d(TAG, "  - ${group.name}: $score")
        }

        val maxScore = scored.maxOf { it.second }
        var filtered = scored.filter { it.second == maxScore }.map { it.first }

        if (filtered.size == 1) {
            Log.d(TAG, "Winner: ${filtered.first().name} (specificity)")
            return filtered.first()
        }

        // 2. 최근 수정 우선
        // TODO: updatedAt 필드 추가 시 사용
        // filtered = filtered.sortedByDescending { it.updatedAt }

        // 3. 생성 순서 우선
        val winner = filtered.maxByOrNull { it.createdAt }!!
        Log.d(TAG, "Winner: ${winner.name} (created at)")

        return winner
    }

    /**
     * 구체성 점수 계산
     * - 7일 모두 활성화: 1점 (매일)
     * - 5일 활성화: 2점 (평일)
     * - 2일 활성화: 2점 (주말)
     * - 그 외: 3점 (특정 요일만)
     */
    private fun calculateSpecificityScore(timeSlots: List<TimeBasedAutoRun>): Int {
        val allDays = timeSlots.flatMap { it.enabledDays }.distinct()

        return when (allDays.size) {
            7 -> 1  // 매일
            5 -> {
                val isWeekday = allDays.containsAll(listOf(
                    DayOfWeek.MONDAY, DayOfWeek.TUESDAY, DayOfWeek.WEDNESDAY,
                    DayOfWeek.THURSDAY, DayOfWeek.FRIDAY
                ))
                if (isWeekday) 2 else 3  // 평일 or 특정 5일
            }
            2 -> {
                val isWeekend = allDays.containsAll(listOf(DayOfWeek.SATURDAY, DayOfWeek.SUNDAY))
                if (isWeekend) 2 else 3  // 주말 or 특정 2일
            }
            else -> 3  // 특정 요일만
        }
    }
}
```

```kotlin
/**
 * 시간대 시작 이벤트 처리 (기반: PRD §4.5.1)
 * 
 * 1. 히스테리시스 체크: 위치 전환이 발생했는지 확인
 * 2. 시간표 활성화 (비위치 스케줄)
 * 3. 알람 등록
 * 4. 집중모드 시작
 */
private suspend fun handleTimeSlotStart(autoRunId: String) {
    // 1. AutoRun 조회
    val autoRun = timeBasedAutoRunDao.getByIdOnce(autoRunId) ?: return
    
    // 2. 스케줄 그룹이 있으면 활성화 여부 확인
    if (autoRun.scheduleGroupId != null) {
        val group = scheduleGroupDao.getByIdOnce(autoRun.scheduleGroupId) ?: return
        if (!group.isActive) {
            Log.i(TAG, "Skipped: Group inactive (${group.name})")
            return
        }
    }
    
    // 3. 집중모드 시작
    startFocusSession(autoRun)
    
    Log.i(TAG, "✅ Time slot started: ${autoRun.hour}:${autoRun.minute}")
}
```

```kotlin
private suspend fun handleTimeSlotEnd(autoRunId: String) {
    // 1. 집중모드 종료
    stopFocusSession(autoRunId)
    
    // 2. 다음 시간대 확인
    val nextAutoRun = findNextTimeSlot()
    if (nextAutoRun != null) {
        Log.d(TAG, "Next time slot: ${nextAutoRun.hour}:${nextAutoRun.minute}")
    } else {
        // 3. 현재 시각/요일/위치 재평가
        reevaluateSchedule()
    }
    
    Log.i(TAG, "✅ Time slot ended")
}
```
