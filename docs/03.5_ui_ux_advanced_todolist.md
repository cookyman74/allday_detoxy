@Composable
fun ScheduleTabScreen(
    onNavigateToDetail: (String) -> Unit,  // 시간표 상세 화면으로 이동
    viewModel: ScheduleGroupViewModel = hiltViewModel()
) {
    val scheduleGroups by viewModel.allScheduleGroups.collectAsState(initial = emptyList())
    val activeScheduleId by viewModel.activeScheduleGroupId.collectAsState()
    
    var showCreateDialog by remember { mutableStateOf(false) }
    
    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("스케줄") },
                colors = TopAppBarDefaults.topAppBarColors(
                    containerColor = MaterialTheme.colorScheme.surface
                )
            )
        },
        floatingActionButton = {
            FloatingActionButton(
                onClick = { showCreateDialog = true }
            ) {
                Icon(Icons.Default.Add, contentDescription = "시간표 추가")
            }
        }
    ) { innerPadding ->
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding)
                .padding(horizontal = 16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
            contentPadding = PaddingValues(vertical = 16.dp)
        ) {
            // 활성 스케줄 카드
            item {
                ActiveScheduleSummaryCard(
                    activeScheduleId = activeScheduleId,
                    scheduleGroups = scheduleGroups
                )
            }
            
            // 다음 예약 카드
            item {
                NextScheduleSummaryCard(
                    scheduleGroups = scheduleGroups
                )
            }
            
            // 섹션 헤더
            item {
                Text(
                    text = "모든 시간표 (${scheduleGroups.size})",
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.Bold,
                    modifier = Modifier.padding(top = 8.dp, bottom = 4.dp)
                )
            }
            
            // 시간표 목록
            items(scheduleGroups) { group ->
                ScheduleSummaryCard(
                    group = group,
                    isActive = group.id == activeScheduleId,
                    onClick = { onNavigateToDetail(group.id) }
                )
            }
            
            // 빈 상태
            if (scheduleGroups.isEmpty()) {
                item {
                    EmptyScheduleCard(
                        onCreateClick = { showCreateDialog = true }
                    )
                }
            }
        }
    }
    
    // 시간표 생성 다이얼로그 (기존 재사용)
    if (showCreateDialog) {
        QuickCreateScheduleDialog(
            onDismiss = { showCreateDialog = false },
            onConfirm = { name, description, iconType, colorHex ->
                viewModel.createScheduleGroup(name, description, iconType, colorHex)
                showCreateDialog = false
            }
        )
    }
}

// 오늘 예정된 다음 스케줄 조회
fun getNextScheduleToday(): Flow<TimeBasedAutoRun?> = flow {
    val now = LocalTime.now()
    val activeGroupId = activeScheduleGroupId.first()
    
    if (activeGroupId != null) {
        repository.getTimeBasedAutoRunsByGroupId(activeGroupId)
            .first()
            .filter { parseTime(it.startTime) > now }
            .sortedBy { parseTime(it.startTime) }
            .firstOrNull()
            ?.let { emit(it) }
    } else {
        emit(null)
    }
}

// 시간표의 시간대 개수 조회
fun getTimeSlotCount(groupId: String): Flow<Int> = 
    repository.getTimeBasedAutoRunsByGroupId(groupId)
        .map { it.size }

// 연결된 위치 이름 조회
fun getLinkedLocationName(groupId: String): Flow<String?> = flow {
    repository.getLocationAutoRunsByScheduleGroupId(groupId)
        .first()
        .firstOrNull()
        ?.let { emit(it.name) }
        ?: emit(null)
}

private fun parseTime(timeString: String): LocalTime {
    // "HH:mm" 형식을 LocalTime으로 파싱
    val parts = timeString.split(":")
    return LocalTime.of(parts[0].toInt(), parts[1].toInt())
}
