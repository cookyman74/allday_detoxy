#### 1.2 주말 제외 로직 (`calculateNextTriggerTime()` 수정)

```kotlin
private fun calculateNextTriggerTime(autoRun: TimeBasedAutoRun): Long? {
    val enabledDays = parseEnabledDays(autoRun.enabledDays)
    if (enabledDays.isEmpty()) {
        return null
    }

    // 🆕 글로벌 옵션: 주말 제외 설정 확인
    val excludeWeekends = runBlocking {
        autoRunSettingsRepository.getExcludeWeekends()
    }

    val now = Calendar.getInstance()
    val targetCalendar = Calendar.getInstance().apply {
        set(Calendar.HOUR_OF_DAY, autoRun.hour)
        set(Calendar.MINUTE, autoRun.minute)
        set(Calendar.SECOND, 0)
        set(Calendar.MILLISECOND, 0)
    }

    for (daysToAdd in 0..13) {
        val checkCalendar = targetCalendar.clone() as Calendar
        checkCalendar.add(Calendar.DAY_OF_YEAR, daysToAdd)
        
        if (checkCalendar.timeInMillis <= now.timeInMillis) {
            continue
        }
        
        val dayOfWeek = getDayOfWeekString(checkCalendar)
        
        // 🆕 주말 제외 옵션 적용
        if (excludeWeekends && (dayOfWeek == "SAT" || dayOfWeek == "SUN")) {
            Log.d(TAG, "⏭️ Skipping weekend: $dayOfWeek (excludeWeekends enabled)")
            continue
        }
        
        if (enabledDays.contains(dayOfWeek)) {
            val excludeInfo = if (excludeWeekends) " [주말 제외 ON]" else ""
            Log.d(TAG, "✅ Next trigger found: ${formatTime(checkCalendar.timeInMillis)} ($dayOfWeek) for ${autoRun.label ?: autoRun.id}$excludeInfo")
            return checkCalendar.timeInMillis
        }
    }

    return null
}
```

**특징**:
- ✅ `excludeWeekends` 설정을 `runBlocking`으로 동기 조회
- ✅ SAT/SUN을 건너뛰기 (주말 제외가 활성화된 경우)
- ✅ 로그에 주말 제외 상태 표시

#### 1.3 사전 알림 로직 (`scheduleTimeBasedAutoRun()` 수정)

```kotlin
fun scheduleTimeBasedAutoRun(autoRun: TimeBasedAutoRun): Boolean {
    if (!autoRun.isEnabled) {
        return false
    }

    val nextTriggerTime = calculateNextTriggerTime(autoRun)
    if (nextTriggerTime == null) {
        return false
    }

    // 🆕 글로벌 옵션: 사전 알림 시간 확인
    val preNotificationMinutes = runBlocking {
        autoRunSettingsRepository.getPreNotificationMinutes()
    }

    // 🆕 사전 알림 알람 등록 (0보다 크면)
    if (preNotificationMinutes > 0) {
        val preNotificationTime = nextTriggerTime - (preNotificationMinutes * 60 * 1000)
        if (preNotificationTime > System.currentTimeMillis()) {
            schedulePreNotification(autoRun, preNotificationTime, preNotificationMinutes)
            Log.d(TAG, "📢 Pre-notification scheduled: ${preNotificationMinutes}분 전 at ${formatTime(preNotificationTime)}")
        }
    }

    // AlarmManager로 스케줄링 시도
    val alarmSuccess = tryScheduleWithAlarmManager(autoRun, nextTriggerTime)
    
    if (!alarmSuccess) {
        return scheduleWithWorkManager(autoRun, nextTriggerTime)
    }
    
    return true
}
```

#### 1.4 사전 알림 알람 스케줄링 메서드 추가

```kotlin
/**
 * 사전 알림 알람 스케줄링
 *
 * 실제 자동 실행 N분 전에 알림을 표시하기 위한 알람입니다.
 * isPreNotification 플래그를 true로 설정하여 Receiver에서 구분할 수 있도록 합니다.
 */
private fun schedulePreNotification(autoRun: TimeBasedAutoRun, preNotificationTime: Long, minutesBefore: Int) {
    val intent = Intent(context, AutoRunAlarmReceiver::class.java).apply {
        action = ACTION_PRE_NOTIFICATION  // 🆕 별도 액션
        putExtra(EXTRA_AUTO_RUN_ID, autoRun.id)
        putExtra(EXTRA_DURATION_MINUTES, autoRun.durationMinutes)
        putExtra(EXTRA_PRESET_TYPE, autoRun.presetType)
        putExtra(EXTRA_LABEL, autoRun.label)
        putExtra(EXTRA_IS_PRE_NOTIFICATION, true)  // 🆕 구분 플래그
    }

    val requestCode = kotlin.math.abs(autoRun.id.hashCode()) + PRE_NOTIFICATION_CODE_OFFSET
    val pendingIntent = PendingIntent.getBroadcast(
        context,
        requestCode,  // 🆕 별도 requestCode (중복 방지)
        intent,
        PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
    )

    try {
        if (canScheduleExactAlarms()) {
            alarmManager.setExactAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                preNotificationTime,
                pendingIntent
            )
        } else {
            alarmManager.setAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                preNotificationTime,
                pendingIntent
            )
        }
        Log.d(TAG, "✅ Pre-notification alarm set for ${autoRun.id}: ${minutesBefore}분 전")
    } catch (e: Exception) {
        Log.e(TAG, "❌ Failed to set pre-notification alarm: ${e.message}")
    }
}
```

### 2. BootCompletedReceiver 구현

**파일**: `app/src/main/java/com/allday/detoxy/receiver/BootCompletedReceiver.kt` (신규, 92줄)

```kotlin
/**
 * 부팅 완료 BroadcastReceiver
 *
 * 기기 재시작 시 모든 활성화된 시간 기반 자동 실행 알람을 재등록합니다.
 *
 * ## 처리 흐름
 * 1. BOOT_COMPLETED 브로드캐스트 수신
 * 2. DB에서 모든 활성화된 자동 실행 조회
 * 3. AutoRunAlarmManager.rescheduleAll() 호출하여 알람 재등록
 */
@AndroidEntryPoint
class BootCompletedReceiver : BroadcastReceiver() {

    @Inject
    lateinit var alarmManager: AutoRunAlarmManager

    @Inject
    lateinit var timeBasedAutoRunDao: TimeBasedAutoRunDao

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    override fun onReceive(context: Context, intent: Intent) {
        if (intent.action != Intent.ACTION_BOOT_COMPLETED) {
            return
        }

        Log.i(TAG, "📱 Device booted, rescheduling all auto-run alarms")

        val pendingResult = goAsync()

        scope.launch {
            try {
                val enabledAutoRuns = timeBasedAutoRunDao.getAll().first().filter { it.isEnabled }
                
                if (enabledAutoRuns.isEmpty()) {
                    Log.d(TAG, "⏭️ No enabled auto-runs to reschedule")
                    return@launch
                }

                Log.d(TAG, "🔄 Rescheduling ${enabledAutoRuns.size} enabled auto-runs")
                
                alarmManager.rescheduleAll(enabledAutoRuns)
                
                Log.i(TAG, "✅ All auto-run alarms rescheduled successfully")
            } catch (e: Exception) {
                Log.e(TAG, "❌ Failed to reschedule alarms: ${e.message}", e)
            } finally {
                pendingResult.finish()
            }
        }
    }
}
```

**특징**:
- ✅ `@AndroidEntryPoint`로 Hilt 의존성 주입 지원
- ✅ `goAsync()`로 비동기 작업 완료 보장 (BroadcastReceiver 10초 제한 대응)
- ✅ 코루틴으로 DB 조회 및 알람 재등록 (Dispatchers.IO)
- ✅ 에러 처리 및 로깅

**AndroidManifest.xml 등록**:

```kotlin
override fun onReceive(context: Context, intent: Intent) {
    val autoRunId = intent.getStringExtra(AutoRunAlarmManager.EXTRA_AUTO_RUN_ID) ?: return
    val durationMinutes = intent.getIntExtra(AutoRunAlarmManager.EXTRA_DURATION_MINUTES, 0)
    val presetType = intent.getStringExtra(AutoRunAlarmManager.EXTRA_PRESET_TYPE)
    val label = intent.getStringExtra(AutoRunAlarmManager.EXTRA_LABEL)

    when (intent.action) {
        AutoRunAlarmManager.ACTION_PRE_NOTIFICATION -> {

        AutoRunAlarmManager.ACTION_AUTO_RUN_ALARM -> {
```

```kotlin
        }
    }

/**
 * 다음 알람 자동 스케줄링
 *
 * DB에서 최신 autoRun 정보를 가져와 다음 발생 시각을 계산하여 알람을 재등록합니다.
 * 주간 반복 알람을 지속적으로 유지하기 위해 필수적입니다.
 */
private suspend fun rescheduleNextAlarm(autoRunId: String) {
    try {
        val autoRunFlow = timeBasedAutoRunDao.getById(autoRunId)
        autoRunFlow.collect { autoRun ->
            if (autoRun == null) {
                Log.w(TAG, "⚠️ AutoRun not found in DB: $autoRunId")
                return@collect
            }

            if (!autoRun.isEnabled) {
                Log.d(TAG, "⏭️ AutoRun is disabled: $autoRunId")
                return@collect
            }

            // 🆕 다음 알람 스케줄링
            val success = alarmManager.scheduleTimeBasedAutoRun(autoRun)
            
            if (success) {
                Log.i(TAG, "✅ Next alarm scheduled for: ${autoRun.label ?: autoRunId}")
            } else {
                Log.e(TAG, "❌ Failed to schedule next alarm")
            }
        }
    } catch (e: Exception) {
        Log.e(TAG, "❌ Error rescheduling alarm: ${e.message}", e)
    }
}
```

**특징**:
- ✅ DB에서 최신 `autoRun` 정보 조회 (Flow.collect)
- ✅ `isEnabled` 체크하여 비활성화된 경우 재스케줄링 건너뛰기
- ✅ `AutoRunAlarmManager.scheduleTimeBasedAutoRun()` 호출로 다음 알람 등록
- ✅ 주간 반복 알람 지속 유지 (월~금 설정 시 매주 자동 반복)
