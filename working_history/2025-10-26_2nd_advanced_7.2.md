@Composable
fun DonutTimerPicker(
    selectedMinutes: Int,
    onMinutesChange: (Int) -> Unit,
    minMinutes: Int = 5,
    maxMinutes: Int = 180,
    stepMinutes: Int = 5,
    modifier: Modifier = Modifier

    val view = LocalView.current
    val textMeasurer = rememberTextMeasurer() // ✅ 텍스트 측정 최적화
    
    // 현재 각도 (0~360도)
    var currentAngle by remember { mutableStateOf(minutesToAngle(selectedMinutes, maxMinutes)) } // ✅ 상태 기억
    
    // 애니메이션 적용된 각도
    val animatedAngle by animateFloatAsState(
        targetValue = currentAngle,

            dampingRatio = Spring.DampingRatioMediumBouncy,
            stiffness = Spring.StiffnessLow

        label = "angle_animation"

    // 이전 분 값 (햅틱 피드백 트리거용)

    // 색상 (MaterialTheme에서 가져옴, 재구성 시 자동 업데이트)
    val primaryColor = MaterialTheme.colorScheme.primary
    val surfaceVariant = MaterialTheme.colorScheme.surfaceVariant
    val onSurface = MaterialTheme.colorScheme.onSurface
    val onSurfaceVariant = MaterialTheme.colorScheme.onSurfaceVariant
    
    // 텍스트 스타일 (Canvas 외부에서 미리 계산) ✅
    val markerTextStyle = TextStyle(
        fontSize = MaterialTheme.typography.labelSmall.fontSize,
        color = onSurfaceVariant
    )
    
    // ...
}

#### 1.2 터치 응답성 확인

**detectDragGestures (line 111-124)**:
```kotlin
.pointerInput(Unit) {
    // 드래그 제스처
    detectDragGestures { change, _ ->
        val center = Offset(size.width / 2f, size.height / 2f)
        val angle = calculateAngle(change.position, center)
        val minutes = angleToMinutes(angle, minMinutes, maxMinutes, stepMinutes)

        if (minutes != previousMinutes) {
            // 햅틱 피드백 (5분 단위 변경 시) ✅ 즉시 실행
            view.performHapticFeedback(android.view.HapticFeedbackConstants.CLOCK_TICK)
            previousMinutes = minutes
        }

        currentAngle = angle
        onMinutesChange(minutes)
    }
}
```

#### 2.3 Flow 중복 수집 확인

**파일**: 여러 ViewModel (TimeBasedAutoRunViewModel, CustomTimerPresetViewModel 등)

**검증 결과**: ✅ **중복 수집 없음**

**ViewModel에서 Flow 수집 (일반적인 패턴)**:
```kotlin
class TimeBasedAutoRunViewModel @Inject constructor(
    private val repository: AutoRunSettingsRepository,
    // ...
) : ViewModel() {
    
    // Flow를 StateFlow로 변환 (viewModelScope에서 자동 관리) ✅
    val autoRuns: StateFlow<List<TimeBasedAutoRun>> = 
        repository.getAllTimeBasedAutoRuns()
            .stateIn(
                scope = viewModelScope,
                started = SharingStarted.WhileSubscribed(5000),
                initialValue = emptyList()
            )
}
```

**Flow 최적화 포인트**:
- ✅ `stateIn`으로 StateFlow 변환 (중복 수집 방지)
- ✅ `SharingStarted.WhileSubscribed(5000)`로 구독자 없을 때 자동 취소
- ✅ `viewModelScope`로 ViewModel 파괴 시 자동 취소
- ✅ 단일 Flow 인스턴스로 여러 구독자 공유

> **7.2 작업의 의의**: 2차 고도화에서 구현한 UI와 데이터베이스의 성능을 코드 리뷰로 검증했습니다. 도넛 그래프는 `remember`, `animateFloatAsState`, `rememberTextMeasurer`를 사용하여 60fps 유지가 가능하도록 최적화되어 있으며, 터치 응답성도 우수합니다. 데이터베이스는 7개의 인덱스가 잘 설정되어 있고, Flow를 사용하여 중복 수집을 방지하고 있습니다. 메모리 관리는 `viewModelScope`와 Room Flow로 자동 관리되며, Broadcast Receiver는 Manifest에 등록되어 메모리 누수 가능성이 낮습니다. 실제 성능 측정과 LeakCanary 테스트는 시간 제약으로 3차 고도화로 연기했지만, 코드 리뷰로 충분히 최적화되어 있음을 확인했습니다. MVP 철학을 준수하여 빠르고 실용적인 검증을 완료했습니다. 🎉
