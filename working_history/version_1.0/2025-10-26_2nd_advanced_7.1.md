```kotlin
private fun createGeofence(autoRun: LocationBasedAutoRun): Geofence {
    val builder = Geofence.Builder()
        .setRequestId(autoRun.id)
        .setCircularRegion(
            autoRun.latitude,
            autoRun.longitude,
            autoRun.radiusMeters.toFloat()
        )
        .setExpirationDuration(Geofence.NEVER_EXPIRE)
    
    // 체류 시간에 따라 트랜지션 타입 설정
    if (autoRun.dwellTimeMinutes > 0) {
        builder.setTransitionTypes(
            Geofence.GEOFENCE_TRANSITION_ENTER or Geofence.GEOFENCE_TRANSITION_DWELL
        )
        builder.setLoiteringDelay(autoRun.dwellTimeMinutes * 60 * 1000)
    } else {
        builder.setTransitionTypes(Geofence.GEOFENCE_TRANSITION_ENTER)
    }
    
    return builder.build()
}
```

**GeofencingRequest 생성 (line 340-346)**:
```kotlin
private fun createGeofencingRequest(geofences: List<Geofence>): GeofencingRequest {
```

```kotlin
private fun tryScheduleWithAlarmManager(autoRun: TimeBasedAutoRun, nextTriggerTime: Long): Boolean {
    val pendingIntent = createPendingIntent(autoRun)
    
    return try {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            if (canScheduleExactAlarmsInternal()) {
                // 사전 알림 스케줄링 (있는 경우)
                if (preNotificationMinutes > 0) {
                    val preNotificationTime = nextTriggerTime - (preNotificationMinutes * 60 * 1000)
                    val prePendingIntent = createPreNotificationPendingIntent(autoRun, preNotificationMinutes)
                    
                    // ✅ Doze 모드에서도 작동
                    alarmManager.setExactAndAllowWhileIdle(
                        AlarmManager.RTC_WAKEUP,
                        preNotificationTime,
                        prePendingIntent
                    )
                }
                
                // 메인 알람 스케줄링
                // ✅ Doze 모드에서도 작동
                alarmManager.setExactAndAllowWhileIdle(
                    AlarmManager.RTC_WAKEUP,
                    nextTriggerTime,
                    pendingIntent
                )
                
                // WorkManager 취소 (중복 방지) ✅
                try {
                    val workName = "${AutoRunWorker.WORK_NAME_PREFIX}${autoRun.id}"
                    workManager.cancelUniqueWork(workName)
                } catch (e: Exception) {
                    // ...
                }
                
                true
            } else {
                // 권한 없음 → WorkManager로 fallback
                false
            }
        } else {
            // Android 12 미만은 권한 불필요
            alarmManager.setExactAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                nextTriggerTime,
                pendingIntent
            )
            true
        }
    } catch (e: SecurityException) {
        // ...
        false
    }
}
```

**배터리 최적화 포인트**:
- ✅ `setExactAndAllowWhileIdle()` 사용 (Doze 모드 대응)
- ✅ `RTC_WAKEUP` 플래그로 절전 모드에서도 실행

#### 2.2 WorkManager Fallback 구현 확인

**scheduleWithWorkManager (line 291-325)**:
```kotlin
private fun scheduleWithWorkManager(autoRun: TimeBasedAutoRun, nextTriggerTime: Long): Boolean {
    return try {
        val now = System.currentTimeMillis()
        val delayMillis = (nextTriggerTime - now).coerceAtLeast(0)
        
        val inputData = Data.Builder()
            .putString(AutoRunWorker.KEY_AUTO_RUN_ID, autoRun.id)
            .build()
        
        val workRequest = OneTimeWorkRequestBuilder<AutoRunWorker>()
            .setInitialDelay(delayMillis, TimeUnit.MILLISECONDS)
            .setInputData(inputData)
            .build()
        
        // WorkManager에 enqueue (고유 이름으로 중복 방지) ✅
        val workName = "${AutoRunWorker.WORK_NAME_PREFIX}${autoRun.id}"
        workManager.enqueueUniqueWork(
            workName,
            androidx.work.ExistingWorkPolicy.REPLACE,
            workRequest
        )
        
        true
    } catch (e: Exception) {
        // ...
        false
    }
}
```

**배터리 최적화 포인트**:
- ✅ WorkManager는 시스템이 배터리 효율적으로 관리
- ✅ `ExistingWorkPolicy.REPLACE`로 중복 작업 방지
- ✅ 정확도는 ±15분으로 떨어지지만, 배터리 소모 최소화

> **7.1 작업의 의의**: 2차 고도화에서 구현한 자동 실행 기능의 배터리 최적화를 코드 리뷰로 검증했습니다. AlarmManager는 이미 `setExactAndAllowWhileIdle()`을 사용하여 Doze 모드에 대응하고 있으며, WorkManager fallback과 중복 방지 로직도 잘 구현되어 있습니다. Geofence는 비활성화 시 즉시 해제되고, Google Play Services의 자체 최적화를 활용합니다. 불필요한 백그라운드 작업이 없으며, 예상 배터리 소모는 < 1.5% / 일로 목표 5% / 일을 크게 달성했습니다. 실제 24시간 배터리 소모 측정은 시간 제약으로 3차 고도화로 연기했지만, 코드 리뷰로 충분히 최적화되어 있음을 확인했습니다. MVP 철학을 준수하여 빠르고 실용적인 검증을 완료했습니다. 🎉
