### 🔧 기술적 세부사항

#### FocusTimer.kt 핵심 코드

```kotlin
enum class FocusState {
    IDLE,       // 대기 상태
    RUNNING,    // 실행 중
    FINISHED,   // 정상 완료
    FAILED      // 포기/실패
}

class FocusTimer(
    private val coroutineScope: CoroutineScope
) {
    private val _state = MutableStateFlow(FocusState.IDLE)
    val state: StateFlow<FocusState> = _state.asStateFlow()

    private val _remainingSeconds = MutableStateFlow(0)
    val remainingSeconds: StateFlow<Int> = _remainingSeconds.asStateFlow()

    fun start(durationMinutes: Int, onFinish: (Boolean) -> Unit = {}) {
        if (_state.value == FocusState.RUNNING) return

        val totalSec = durationMinutes * 60
        _totalSeconds.value = totalSec
        _remainingSeconds.value = totalSec
        _state.value = FocusState.RUNNING

        timerJob = coroutineScope.launch {
            try {
                while (_remainingSeconds.value > 0 && _state.value == FocusState.RUNNING) {
                    delay(1000) // 1초 대기
                    _remainingSeconds.value -= 1
                }

                if (_state.value == FocusState.RUNNING && _remainingSeconds.value == 0) {
                    _state.value = FocusState.FINISHED
                    onFinish(true)
                }
            } catch (e: CancellationException) {
                throw e
            }
        }
    }

    fun giveUp() {
        if (_state.value != FocusState.RUNNING) return
        timerJob?.cancel()
        _state.value = FocusState.FAILED
    }

    fun reset() {
        timerJob?.cancel()
        _state.value = FocusState.IDLE
        _remainingSeconds.value = 0
        _totalSeconds.value = 0
    }

    fun getProgress(): Float {
        if (_totalSeconds.value == 0) return 0f
        val elapsed = _totalSeconds.value - _remainingSeconds.value
        return elapsed.toFloat() / _totalSeconds.value.toFloat()
    }

    fun getFormattedTime(): String {
        val minutes = _remainingSeconds.value / 60
        val seconds = _remainingSeconds.value % 60
        return String.format("%02d:%02d", minutes, seconds)
    }
}
```

#### TimerViewModel.kt 핵심 코드

```kotlin
@HiltViewModel
class TimerViewModel @Inject constructor() : ViewModel() {

    private val focusTimer = FocusTimer(viewModelScope)

    val timerState: StateFlow<FocusState> = focusTimer.state
    val remainingSeconds: StateFlow<Int> = focusTimer.remainingSeconds
    val totalSeconds: StateFlow<Int> = focusTimer.totalSeconds

    val presetDurations = listOf(25, 45, 60)

    fun startTimer(durationMinutes: Int) {
        // AccessibilityService 활성화
        FocusAccessibilityService.isTimerRunning = true

        // 타이머 시작
        focusTimer.start(durationMinutes) { success ->
            onTimerFinish(success)
        }
    }

    fun giveUpTimer() {
        FocusAccessibilityService.isTimerRunning = false
        focusTimer.giveUp()
    }

    fun resetTimer() {
        FocusAccessibilityService.isTimerRunning = false
        focusTimer.reset()
    }

    private fun onTimerFinish(success: Boolean) {
        FocusAccessibilityService.isTimerRunning = false

        // TODO: Week 3 - 포인트 지급 및 스트릭 업데이트
    }

    override fun onCleared() {
        super.onCleared()
        if (timerState.value == FocusState.RUNNING) {
            resetTimer()
```

#### TimerScreen.kt UI 구조

```kotlin
@Composable
fun TimerScreen(viewModel: TimerViewModel = viewModel()) {
    val timerState by viewModel.timerState.collectAsState()

    Column {
        Text("집중 타이머", style = MaterialTheme.typography.headlineMedium)

        // 원형 프로그레스 바 & 타이머 표시
        CircularTimerDisplay(
            state = timerState,
            formattedTime = viewModel.getFormattedTime(),
            progress = viewModel.getProgress()
        )

        // 타이머 상태에 따른 UI
        when (timerState) {
            FocusState.IDLE -> {
                PresetButtons(
                    presets = viewModel.presetDurations,
                    onPresetClick = { viewModel.startTimer(it) }
                )
            }
            FocusState.RUNNING -> {
                Button(onClick = { viewModel.giveUpTimer() }) {
                    Text("포기하기")
                }
            }
            FocusState.FINISHED -> {
                Text("🎉 타이머 완료!")
                Button(onClick = { viewModel.resetTimer() }) {
                    Text("새 타이머 시작")
                }
            }
            FocusState.FAILED -> {
                Text("타이머 포기")
                Button(onClick = { viewModel.resetTimer() }) {
                    Text("다시 시작")
                }
            }
        }
    }
}
```
