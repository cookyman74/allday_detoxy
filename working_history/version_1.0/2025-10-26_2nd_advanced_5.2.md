##### 3.2.2 deletePreset

```kotlin
fun deletePreset(presetId: String) {
    viewModelScope.launch {
        repository.delete(presetId)
        
        // 삭제 후 표시 순서 재정렬
        reorderPresets()
    }
}

private suspend fun reorderPresets() {
    val currentPresets = repository.getAll().stateIn(viewModelScope).value
    currentPresets.forEachIndexed { index, preset ->
        repository.updateDisplayOrder(preset.id, index)
    }
}
```

**특징**:
- ✅ 삭제 후 자동 순서 재정렬
- ✅ displayOrder 일관성 보장
- ✅ 드래그 앤 드롭 준비 (향후 구현)

```kotlin
// CustomTimerPresetRepository 주입
private val presetRepository: CustomTimerPresetRepository

// 커스텀 프리셋 목록 노출
val customPresets: StateFlow<List<CustomTimerPreset>> = presetRepository.getAll()
    .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

// 프리셋 관리 메서드
fun saveCustomPreset(name: String, durationMinutes: Int, presetType: String?) {
    viewModelScope.launch {
        presetRepository.insert(
            CustomTimerPreset(
                name = name,
                durationMinutes = durationMinutes,
                presetType = presetType,
                displayOrder = presetRepository.getCount()
            )
        )
    }
}

fun deleteCustomPreset(presetId: String) {
    viewModelScope.launch {
        presetRepository.delete(presetId)
    }
}

fun updateCustomPreset(preset: CustomTimerPreset) {
    viewModelScope.launch {
        presetRepository.update(preset)
    }
}

fun incrementPresetUsage(presetId: String?) {
    if (presetId != null) {
        viewModelScope.launch {
            presetRepository.incrementUsageCount(presetId)
        }
    }
}
```

```kotlin
// SavePresetDialog
if (showSaveDialog) {
    SavePresetDialog(
        durationMinutes = selectedMinutes,

        },

// EditPresetDialog
if (showEditDialog && selectedPreset != null) {
    EditPresetDialog(
        presetName = selectedPreset!!.name,
        presetType = selectedPreset!!.presetType,
        onSave = { newName, newType ->
            val updated = selectedPreset!!.copy(name = newName, presetType = newType)
            viewModel.updateCustomPreset(updated)
            selectedPreset = null
        },
        onDismiss = {
            showEditDialog = false
            selectedPreset = null
        }
    )
}

// DeletePresetDialog
if (showDeleteDialog && selectedPreset != null) {
    DeletePresetDialog(
        presetName = selectedPreset!!.name,
        onConfirm = {
            viewModel.deleteCustomPreset(selectedPreset!!.id)
            selectedPreset = null
        },
        onDismiss = {
            showDeleteDialog = false
            selectedPreset = null
        }
    )
}

// PresetManagementBottomSheet
if (showPresetSheet && selectedPreset != null) {
    PresetManagementBottomSheet(
        preset = selectedPreset!!,
        onEdit = {
            showPresetSheet = false
            showEditDialog = true
        },
        onDelete = {
            showPresetSheet = false
            showDeleteDialog = true
        },
        onDismiss = {
            showPresetSheet = false
            selectedPreset = null
        }
    )
```
