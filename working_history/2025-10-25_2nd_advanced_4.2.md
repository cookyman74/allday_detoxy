**체크 항목**:
```kotlin
suspend fun addGeofence(autoRun: LocationBasedAutoRun): Result<Unit> {
    // 1. Play Services 체크
    if (!isPlayServicesAvailable()) {
        return Result.failure(
            GeofenceException("Google Play Services가 설치되어 있지 않습니다.")
        )
    }
    
    // 2. 위치 서비스 활성화 체크
    if (!isLocationEnabled()) {
        return Result.failure(
            GeofenceException("위치 서비스가 꺼져 있습니다. 설정에서 위치 서비스를 켜주세요.")
        )
    }
    
    // 3. 위치 권한 체크
    if (!hasLocationPermission()) {
        return Result.failure(
            GeofenceException("위치 권한이 없습니다. 앱 설정에서 위치 권한을 허용해주세요.")
        )
    }
    
    // 4. 백그라운드 위치 권한 체크 (Android 10+)
    if (!hasBackgroundLocationPermission()) {
        return Result.failure(
            GeofenceException("백그라운드 위치 권한이 없습니다. '항상 허용'을 선택해주세요.")
        )
    }
    
    // 5. 최대 개수 체크
    val otherEnabledCount = locationBasedAutoRunDao.getEnabledCountExcept(autoRun.id)
    if (otherEnabledCount >= MAX_GEOFENCES) {
        return Result.failure(
            GeofenceException("최대 ${MAX_GEOFENCES}개까지만 등록할 수 있습니다.")
        )
    }
}
```

**특징**:
- 모든 사전 조건을 순차적으로 체크
- 각 실패 조건마다 명확한 에러 메시지 제공
- `Result<Unit>` 타입으로 성공/실패 명확히 구분

#### 1.2 Geofence 생성 및 등록

```kotlin
// Geofence 생성
private fun createGeofence(autoRun: LocationBasedAutoRun): Geofence {
    return Geofence.Builder()
        .setRequestId(autoRun.id)
        .setCircularRegion(
            autoRun.latitude,
            autoRun.longitude,
            autoRun.radiusMeters.toFloat()
        )
        .setExpirationDuration(Geofence.NEVER_EXPIRE)
        .setTransitionTypes(Geofence.GEOFENCE_TRANSITION_ENTER)
        .setLoiteringDelay(autoRun.dwellTimeMinutes * 60 * 1000)  // 체류 시간
        .build()
}

// GeofencingRequest 생성
private fun createGeofencingRequest(geofences: List<Geofence>): GeofencingRequest {
    return GeofencingRequest.Builder()
        .setInitialTrigger(GeofencingRequest.INITIAL_TRIGGER_ENTER)
        .addGeofences(geofences)
        .build()
}

// PendingIntent 생성
private fun createPendingIntent(autoRun: LocationBasedAutoRun): PendingIntent {
    val intent = Intent(context, GeofenceTransitionsReceiver::class.java).apply {
        action = ACTION_GEOFENCE_TRANSITION
        putExtra(EXTRA_LOCATION_ID, autoRun.id)
        putExtra(EXTRA_LOCATION_LABEL, autoRun.label)
        putExtra(EXTRA_DURATION_MINUTES, autoRun.durationMinutes)
        putExtra(EXTRA_PRESET_TYPE, autoRun.presetType)
        putExtra(EXTRA_REQUIRES_CONFIRMATION, autoRun.requiresUserConfirmation)
        putExtra(EXTRA_DWELL_TIME_MINUTES, autoRun.dwellTimeMinutes)
    }
    
    return PendingIntent.getBroadcast(
        context,
        autoRun.id.hashCode(),
        intent,
        PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
    )
}
```

**특징**:
- 체류 시간(dwellTimeMinutes) 지원: 0/1/3/5분
- ENTER 트리거만 사용 (진입 시 자동 실행)
- NEVER_EXPIRE: 영구 유효
- 사용자 확인 필요 여부(requiresUserConfirmation) 전달

### 2. ViewModel에서 Geofence 호출

**파일**: `app/src/main/java/com/allday/detoxy/presentation/viewmodel/LocationBasedAutoRunViewModel.kt`

#### 2.1 addLocation (위치 추가)

```kotlin
fun addLocation(location: LocationBasedAutoRun) {
    viewModelScope.launch {
        try {
            // 1. Geofence 등록 먼저 시도 (활성화된 경우만)
            if (location.isEnabled) {
                val result = geofenceManager.addGeofence(location)
                if (result.isFailure) {
                    val exception = result.exceptionOrNull()
                    _errorState.value = LocationError.GeofenceError(
                        exception?.message ?: "Geofence 등록 실패"
                    )
                    return@launch  // 실패 시 DB 저장 안 함
                }
            }

            // 2. Geofence 성공 후 DB 저장
            repository.insert(location)

        } catch (e: Exception) {
            // 3. DB 저장 실패 시 Geofence 롤백
            if (location.isEnabled) {
                geofenceManager.removeGeofence(location.id)
            }
            _errorState.value = LocationError.DatabaseError("저장 실패")
        }
    }
}
```

**트랜잭션 순서**:
1. Geofence 등록 먼저
2. 성공 시 DB 저장
3. 실패 시 Geofence 롤백

#### 2.2 updateLocation (위치 수정)

```kotlin
fun updateLocation(location: LocationBasedAutoRun) {
    viewModelScope.launch {
        try {
            // 1. 기존 Geofence 제거
            geofenceManager.removeGeofence(location.id)

            // 2. 새 Geofence 등록 (활성화된 경우만)
            if (location.isEnabled) {
                val result = geofenceManager.addGeofence(location)
                if (result.isFailure) {
                    _errorState.value = LocationError.GeofenceError("등록 실패")
                    return@launch
                }
            }

            // 3. DB 업데이트
            repository.update(location)

        } catch (e: Exception) {
            // DB 업데이트 실패 → Geofence 제거
            geofenceManager.removeGeofence(location.id)
            _errorState.value = LocationError.DatabaseError("업데이트 실패")
        }
    }
}
```

#### 2.3 toggleLocation (활성화 토글)

```kotlin
fun toggleLocation(locationId: String, isEnabled: Boolean) {
    viewModelScope.launch {
        try {
            val location = repository.getById(locationId).first() ?: return@launch
            
            if (isEnabled) {
                // 활성화: Geofence 등록
                val result = geofenceManager.addGeofence(location)
                if (result.isFailure) {
                    _errorState.value = LocationError.GeofenceError("등록 실패")
                    return@launch
                }
            } else {
                // 비활성화: Geofence 제거
                geofenceManager.removeGeofence(locationId)
            }
            
            // DB 상태 업데이트
            repository.toggleEnabled(locationId, isEnabled)

        } catch (e: Exception) {
            // 롤백 처리
            if (isEnabled) {
                geofenceManager.removeGeofence(locationId)
            }
            _errorState.value = LocationError.DatabaseError("토글 실패")
        }
    }
}
```

```kotlin
fun hasLocationPermission(): Boolean {
    return ContextCompat.checkSelfPermission(
        context,
        Manifest.permission.ACCESS_FINE_LOCATION
    ) == PackageManager.PERMISSION_GRANTED
}

fun hasBackgroundLocationPermission(): Boolean {
    return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
        ContextCompat.checkSelfPermission(
            context,
            Manifest.permission.ACCESS_BACKGROUND_LOCATION
        ) == PackageManager.PERMISSION_GRANTED
    } else {
        true  // Android 10 미만은 백그라운드 권한 불필요
    }
}
```

```kotlin
/**
 * 모든 활성화된 Geofence 재등록
 *
 * 앱 재시작(BOOT_COMPLETED) 시 호출되어 모든 활성화된 위치 기반 자동 실행의 Geofence를 재등록합니다.
 *
 * 처리 방식:
 * - 사전 조건 체크 (Play Services, 위치 서비스, 권한) 수행
 * - 실패한 항목은 로그에 기록하고 계속 진행 (일부 실패해도 나머지 등록)
 *
 * @param enabledLocations 활성화된 위치 기반 자동 실행 리스트
 */
suspend fun rescheduleAll(enabledLocations: List<LocationBasedAutoRun>) {
    if (enabledLocations.isEmpty()) {
        Log.i(TAG, "ℹ️ No enabled locations to reschedule")
        return
    }
    
    Log.i(TAG, "🔄 Rescheduling ${enabledLocations.size} geofences")
    
    var successCount = 0
    var failureCount = 0
    
    enabledLocations.forEach { location ->
        val result = addGeofence(location)
        if (result.isSuccess) {
            successCount++
            Log.d(TAG, "✅ Geofence rescheduled: ${location.label} (ID: ${location.id})")
        } else {
            failureCount++
            val error = result.exceptionOrNull()
            Log.w(TAG, "⚠️ Failed to reschedule geofence for ${location.label}: ${error?.message}")
        }
    }
    
    Log.i(TAG, """
        📊 Geofence reschedule completed:
        - Total: ${enabledLocations.size}
        - Success: $successCount
        - Failure: $failureCount
    """.trimIndent())
}
```
