```kotlin
object OverlayDisplayRules {

    const val MAX_GOAL_TEXT_LENGTH = 30       // 목표 텍스트 최대 길이

enum class OverlayDisplayMode {
    EMOJI_ONLY,      // 🎯 이모지만 표시
    HIDDEN,          // 모두 숨김
    GOAL_ONLY,       // 목표만 표시
    GOAL_AND_TODOS   // 목표 + 할일 표시
}
```
